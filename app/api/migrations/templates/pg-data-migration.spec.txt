import { testingEnvironment } from '#api/utils/testingEnvironment.js';
import { migration } from '../{{nextSchemaDelta}}-{{dashName name}}.js';

const TENANT_A = 'tenant-a';
const TENANT_B = 'tenant-b';

/** The tables the migration touches: read raw, every tenant, to assert on. */
const TABLES: string[] = [];

const run = async () => testingEnvironment.pg.runDataMigration(migration, [TENANT_A, TENANT_B]);

/** Every row of TABLES, order-insensitive: an UPDATE moves rows around on disk. */
const snapshot = async () =>
  Object.fromEntries(
    await Promise.all(
      TABLES.map(async table => [
        table,
        (await testingEnvironment.pg.getAllFrom(table)).map(row => JSON.stringify(row)).sort(),
      ])
    )
  );

describe('{{nextSchemaDelta}}-{{dashName name}}', () => {
  beforeAll(async () => {
    await testingEnvironment.setUp({}, { postgres: true });
  });

  afterAll(async () => {
    await testingEnvironment.tearDown();
  });

  beforeEach(async () => {
    // Rows in the old shape, for both tenants: { table: [{ tenant_id: TENANT_A, ... }, ...] }
    await testingEnvironment.setFixtures({}, {});
  });

  it('should convert every tenant rows to the new shape, and nothing else', async () => {
    await run();
    expect(await snapshot()).toBe('asserted on the raw tables, for both tenants');
  });

  it('should give the same result when run twice', async () => {
    expect(TABLES).not.toEqual([]);
    await run();
    const once = await snapshot();

    await run();

    expect(await snapshot()).toEqual(once);
  });

  it('should do nothing on empty tables', async () => {
    expect(TABLES).not.toEqual([]);
    await testingEnvironment.setFixtures({}, Object.fromEntries(TABLES.map(table => [table, []])));

    await run();

    expect(await snapshot()).toEqual(Object.fromEntries(TABLES.map(table => [table, []])));
  });

  it('should check if a reindex is needed', async () => {
    const { reindexed } = await run();
    expect(migration.reindex).toBe(undefined);
    expect(reindexed).toEqual([]);
  });
});
